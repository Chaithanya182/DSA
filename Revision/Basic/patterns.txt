class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        //2*n-1
        for(int i=1;i <=n;i++)
            {
               //spaces 
                for(int j=0;j<n-i;j++)
                    {
                        System.out.print(" ");
                    }
                //stars
                for(int k=1 ; k <=(2*(i-1)) + 1 ; k++)
                    {
                        System.out.print("*");
                    }
                System.out.println();
            }
    }
}

import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        //2*n-1
        for(int i=0;i<2*n-1;i++)
            {
            for(int j = 0;j<2*n-1 ; j++)
                {
                    int left = i;
                    int right = 2*n -2- i;
                    int top = j;
                    int bottom = 2*n-2-j;
                    int min = Math.min(Math.min(left,right) , Math.min(top,bottom));

                    System.out.print(n-min);
                }
                System.out.println();

            }
    }
